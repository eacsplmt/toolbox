# 一、电源拉满
# ============================================
Write-Host "极致性能电源方案" -ForegroundColor Gray

if ($perfLine) {
    $guid = ($perfLine -split '\s+')[3]
    powercfg -setactive $guid
    Write-Host "  卓越性能已启用" -ForegroundColor Green
} else {
    Write-Host "  未找到卓越性能，添加并启用..." -ForegroundColor Yellow
    $result = powercfg -duplicatescheme e9a42b02-d5df-448d-aa00-03f14749eb61 2>&1
    $perfLine = powercfg /l | Select-String -Pattern "卓越性能|Ultimate Performance"
    if ($perfLine) {
        $guid = ($perfLine -split '\s+')[3]
        powercfg -setactive $guid
        Write-Host "  卓越性能已启用" -ForegroundColor Green
    } else {
        Write-Host "  添加失败，手动检查" -ForegroundColor Red
    }
}
